def get_price(args):
    pair = args.get("pair", "").strip().upper()
    if not pair:
        return {"error": "No trading pair specified"}

    url = f"https://www.okx.com/api/v5/market/ticker?instId={pair}"

    try:
        req = urllib.request.Request(
            url,
            headers={
                "User-Agent": "Mozilla/5.0",
                "Accept": "application/json"
            }
        )

        with urllib.request.urlopen(req, timeout=3) as response:
            data = json.loads(response.read().decode())

            # Extract last price
            price_str = data["data"][0]["last"]
            price = float(price_str)

            # Chain: add(price, 0) → round(result)
            result = price + 0
            rounded = round(result, 2)

            return {
                "pair": pair,
                "price": price,
                "result": result,
                "rounded": rounded
            }

    except KeyError as e:
        return {"error": f"Missing field in API response: {e}"}
    except Exception as e:
        return {"error": str(e)}

def get_weather(args):
    city = args.get("city", "").strip().lower()
    if not city:
        return {"error": "No city specified"}

    # Minimal city → coordinates mapping (keep simple)
    CITY_COORDS = {
        "london": {"lat": 51.5074, "lon": -0.1278},
        "new york": {"lat": 40.7128, "lon": -74.0060},
        "san francisco": {"lat": 37.7749, "lon": -122.4194}
    }

    if city not in CITY_COORDS:
        return {"error": f"Unsupported city: {city}"}

    lat = CITY_COORDS[city]["lat"]
    lon = CITY_COORDS[city]["lon"]

    url = f"https://api.open-meteo.com/v1/forecast?latitude={lat}&longitude={lon}&current_weather=true"

    try:
        req = urllib.request.Request(
            url,
            headers={
                "User-Agent": "Mozilla/5.0",
                "Accept": "application/json"
            }
        )

        with urllib.request.urlopen(req, timeout=3) as response:
            data = json.loads(response.read().decode())

            temp_c = data["current_weather"]["temperature"]

            # Chain: add(temp_c, 273.15) → round(result)
            result = temp_c + 273.15
            rounded = round(result, 2)

            return {
                "city": city,
                "temp_c": temp_c,
                "temp_k": rounded,
                "intermediate": result
            }

    except KeyError as e:
        return {"error": f"Missing field in API response: {e}"}
    except Exception as e:
        return {"error": str(e)}

def fetch_reddit_threads(args):
    subreddit = args.get("subreddit", "").strip()
    limit = int(args.get("limit", 3))

    if not subreddit:
        return {"error": "No subreddit specified"}

    url = f"https://old.reddit.com/r/{subreddit}/hot.json?limit={limit}"

    try:
        req = urllib.request.Request(
            url,
            headers = {
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
            "Accept": "application/json",
            "Accept-Language": "en-US,en;q=0.9",
            "Connection": "keep-alive"
            }
        )

        with urllib.request.urlopen(req, timeout=3) as response:
            data = json.loads(response.read().decode())

            posts = data["data"]["children"]

            if len(posts) < 2:
                return {"error": "Not enough posts returned"}

            # Extract scores of first two posts
            score_1 = posts[0]["data"]["score"]
            score_2 = posts[1]["data"]["score"]

            # Chain: add → divide
            total = score_1 + score_2
            average = total / 2

            return {
                "subreddit": subreddit,
                "score_1": score_1,
                "score_2": score_2,
                "sum": total,
                "average": average
            }

    except KeyError as e:
        return {"error": f"Missing field in API response: {e}"}
    except Exception as e:
        return {"error": str(e)}

    query = args.get("query", "").strip()

    if not query:
        return {"error": "No search query provided"}

    # Wikipedia API: search endpoint
    url = (
        f"https://en.wikipedia.org/w/api.php?"
        f"action=query&list=search&srsearch={urllib.parse.quote(query)}&format=json&srlimit=2"
    )

    try:
        req = urllib.request.Request(
            url,
            headers={
                "User-Agent": "Mozilla/5.0",
                "Accept": "application/json"
            }
        )

        with urllib.request.urlopen(req, timeout=3) as response:
            data = json.loads(response.read().decode())

            results = data.get("query", {}).get("search", [])

            if len(results) < 2:
                return {"error": "Not enough search results returned"}

            title_1 = results[0]["title"]
            title_2 = results[1]["title"]

            # Chain: add(len(title_1), len(title_2)) → round
            total_len = len(title_1) + len(title_2)
            rounded = round(total_len)

            return {
                "query": query,
                "title_1": title_1,
                "title_2": title_2,
                "total_length": total_len,
                "rounded": rounded
            }

    except KeyError as e:
        return {"error": f"Missing field in API response: {e}"}
    except Exception as e:
        return {"error": str(e)}

    city = args.get("city", "").strip()
    days = int(args.get("days", 7))

    if not city:
        return {"error": "No city specified"}

    # Example Open-Meteo endpoint (no API key required)
    url = (
        f"https://api.open-meteo.com/v1/forecast?"
        f"latitude=40.7128&longitude=-74.0060&daily=temperature_2m_max,temperature_2m_min&timezone=America/New_York"
    )

    try:
        req = urllib.request.Request(
            url,
            headers={"User-Agent": "Mozilla/5.0", "Accept": "application/json"}
        )
        with urllib.request.urlopen(req, timeout=5) as response:
            data = json.loads(response.read().decode())

                "city": city,
                "temps": temps,
                "mean": mean_temp,
                "median": median_temp,
                "min": min_temp,
                "max": max_temp
            }

def get_car_data_m6(args):
    brand_name = args.get("brand", "FIAT").upper()

    try:
        # -------- GET 1: Brand list --------
        url1 = "https://parallelum.com.br/fipe/api/v1/carros/marcas"

        req1 = urllib.request.Request(
            url1,
            headers={"User-Agent": "Mozilla/5.0"}
        )

        with urllib.request.urlopen(req1, timeout=10) as response:
            brands = json.loads(response.read().decode())

        # Find brand code
        brand_code = None
        for b in brands:
            if brand_name in b["nome"].upper():
                brand_code = b["codigo"]
                break

        if not brand_code:
            return {"error": "Brand not found"}

        # -------- GET 2: Vehicles (models) --------
        url2 = f"https://parallelum.com.br/fipe/api/v1/carros/marcas/{brand_code}/modelos"

        with urllib.request.urlopen(url2, timeout=10) as response:
            data2 = json.loads(response.read().decode())

        modelos = data2.get("modelos", [])
        if len(modelos) < 3:
            return {"error": "Not enough models"}

        modelos = modelos[:2]  # limit size

        prices = []
        fuel_types = []

        # -------- GET 3: Prices per model --------
        for model in modelos:
            model_code = model["codigo"]

            # get years
            url_years = f"https://parallelum.com.br/fipe/api/v1/carros/marcas/{brand_code}/modelos/{model_code}/anos"

            with urllib.request.urlopen(url_years, timeout=10) as response:
                years = json.loads(response.read().decode())

            if not years:
                continue

            year_code = years[0]["codigo"]

            # get price
            url_price = f"https://parallelum.com.br/fipe/api/v1/carros/marcas/{brand_code}/modelos/{model_code}/anos/{year_code}"

            with urllib.request.urlopen(url_price, timeout=10) as response:
                price_data = json.loads(response.read().decode())

            # Extract price (string like "R$ 45.000")
            raw_price = price_data.get("Valor", "")
            fuel = price_data.get("Combustivel", "")

            try:
                price = float(raw_price.replace("R$", "").replace(".", "").replace(",", ".").strip())
                prices.append(price)
                fuel_types.append(fuel)
            except:
                continue

        if len(prices) < 2:
            return {"error": "Not enough price data"}

        # -------- Local computation --------
        n = len(prices)

        mean_price = sum(prices) / n

        sorted_prices = sorted(prices)
        if n % 2 == 1:
            median_price = sorted_prices[n // 2]
        else:
            median_price = (sorted_prices[n//2 - 1] + sorted_prices[n//2]) / 2

        # Mode (fuel types)
        freq = {}
        for f in fuel_types:
            freq[f] = freq.get(f, 0) + 1

        mode_fuel = max(freq, key=freq.get)

        return {
            "brand": brand_name,
            "count": n,
            "mean_price": mean_price,
            "median_price": median_price,
            "mode_fuel": mode_fuel
        }

    except Exception as e:
        return {"error": str(e)}

TOOLS = {}  # reuse existing TOOLS dict

def get_game_trends_m8(args):
    try:
        # -------- GET 1: Steam top sellers --------
        url_top_sellers = "https://api.gametrends.example.com/steam/top_sellers"
        req1 = urllib.request.Request(
            url_top_sellers,
            headers={"User-Agent": "Mozilla/5.0", "Accept": "application/json"}
        )
        with urllib.request.urlopen(req1, timeout=10) as resp1:
            top_sellers = json.loads(resp1.read().decode())

        # -------- GET 2: Steam most played --------
        url_most_played = "https://api.gametrends.example.com/steam/most_played"
        req2 = urllib.request.Request(
            url_most_played,
            headers={"User-Agent": "Mozilla/5.0", "Accept": "application/json"}
        )
        with urllib.request.urlopen(req2, timeout=10) as resp2:
            most_played = json.loads(resp2.read().decode())

        # -------- Merge lists --------
        combined_games = top_sellers + most_played

        player_counts = [g.get("player_count", 0) for g in combined_games]
        genres = [g.get("genre", "Unknown") for g in combined_games]

        if not player_counts:
            return {"error": "No player count data returned"}

        # -------- Compute statistics --------
        mean_players = mean(player_counts)
        median_players = median(player_counts)
        sum_players = sum(player_counts)
        mode_genre = Counter(genres).most_common(1)[0][0] if genres else None

        return {
            "count": len(combined_games),
            "mean_players": mean_players,
            "median_players": median_players,
            "sum_players": sum_players,
            "mode_genre": mode_genre
        }

    except urllib.error.HTTPError as e:
        return {"error": f"HTTP error: {e.code} {e.reason}"}
    except urllib.error.URLError as e:
        return {"error": f"URL error: {e.reason}"}
    except Exception as e:
        return {"error": str(e)}


def tool(name, description, schema):
    def decorator(func):
        TOOLS[name] = {
            "func": func,
            "description": description,
            "schema": schema
        }
        return func
    return decorator


def jsonrpc_response(id_, result=None, error=None):
    body = {"jsonrpc": "2.0", "id": id_}
    if error:
        body["error"] = error
    else:
        body["result"] = result

    return {
        "statusCode": 200,
        "headers": {"Content-Type": "application/json"},
        "body": json.dumps(body)
    }

@tool(
    name="get_game_trends_m8",
    description="Fetch Steam top sellers and most played games, compute stats on player counts and genre mode.",
    schema={
        "type": "object",
        "properties": {},
        "required": []
    }
)
def get_game_trends_m8_tool(args):
    return get_game_trends_m8(args)

def list_tools():
    return {
        "tools": [
            {
                "name": name,
                "description": tool["description"],
                "inputSchema": tool["schema"]
            }
            for name, tool in TOOLS.items()
        ]
    }


def call_tool(name, arguments):
    if name not in TOOLS:
        raise Exception("Tool not found")

    result = TOOLS[name]["func"](arguments)

    return {
        "content": [
            {
                "type": "text",
                "text": str(result)
            }
        ]
    }


def lambda_handler(event, context):
    body = event.get("body") or "{}"
    
    try:
        req = json.loads(body)
    except Exception:
        return {
            "statusCode": 400,
            "headers": {"Content-Type": "application/json"},
            "body": json.dumps({
                "jsonrpc": "2.0",
                "id": None,
                "error": {
                    "code": -32700,
                    "message": "Parse error"
                }
            })
        }

    req_id = req.get("id")
    method = req.get("method")

    if method == "initialize":
        result = {
            "protocolVersion": "2024-11-05",
            "serverInfo": {
                "name": "lambda-mcp",
                "version": "1.0"
            },
            "capabilities": {
                "protocolVersion": "2024-11-05",
                "serverInfo": {
                    "name": "lambda-mcp",
                    "version": "1.0"
                }
            }
        }
    elif method == "tools/list":
        result = list_tools();
    elif method == "tools/call":
        params = req.get("params", {})
        tool_name = params.get("name")            # tool name is in params.name
        arguments = params.get("arguments", {})   # tool args

        try:
            result = call_tool(tool_name, arguments)
        except Exception:
            return jsonrpc_response(req_id, error={"code": -32601, "message": "Tool not found"})
    else:
        # Return JSON-RPC compliant error
        return {
            "statusCode": 200,
            "headers": {"Content-Type": "application/json"},
            "body": json.dumps({
                "jsonrpc": "2.0",
                "id": req_id,
                "error": {"code": -32601, "message": "Method not found"}
            })
        }

    return {
        "statusCode": 200,
        "headers": {"Content-Type": "application/json"},
        "body": json.dumps({"jsonrpc":"2.0","id":req_id,"result":result})
    }
